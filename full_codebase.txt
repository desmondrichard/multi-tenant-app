var RDB *redis.Client

func ConnectRedis() {
	addr := os.Getenv("REDIS_ADDR")
	password := os.Getenv("REDIS_PASSWORD")

	// If no address is provided, skip gracefully (useful for quick local tests)
	if addr == "" {
		log.Println("âš ï¸ Redis not configured, skipping connection")
		return
	}

	// Upstash requires TLS and a password
	RDB = redis.NewClient(&redis.Options{
		Addr:     addr,
		Password: password,
		TLSConfig: &tls.Config{
			MinVersion: tls.VersionTLS12,
		},
	})

	if err := RDB.Ping(context.Background()).Err(); err != nil {
		// We use log.Println instead of log.Fatal here so that if Redis goes down,
		// your whole app doesn't crash. It will just log a warning.
		log.Println("âš ï¸ Redis connect error: ", err)
		return
	}

	log.Println("âœ… Redis connected to Upstash")
}

// GenerateTask creates a task based on a prompt
func (a *AIController) GenerateTask(c *gin.Context) {
	var in AITaskRequest
	if err := c.ShouldBindJSON(&in); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
		return
	}

	orgID, _ := primitive.ObjectIDFromHex(c.GetString("orgId"))
	userID, _ := primitive.ObjectIDFromHex(c.GetString("userId"))
	projectID, _ := primitive.ObjectIDFromHex(in.ProjectID)

	// 1. Fetch users for AI context
	users, _ := a.UserRepo.ListByOrg(c.Request.Context(), orgID)
	var userNames []string
	for _, u := range users {
		userNames = append(userNames, u.Name)
	}

	// 2. Build Prompt
	prompt := fmt.Sprintf(`
		You are an AI assistant for a project management app.
		The user will give you a prompt. You must extract:
		- "title": A short title for the task.
		- "priority": Must be exactly "low", "medium", or "high".
		- "assignee": The name of the person to assign. Choose from: %v. If no one matches, leave it empty.
		
		Respond ONLY with a valid JSON object. Do not include markdown formatting.
		User Prompt: "%s"
	`, userNames, in.Prompt)

	// 3. Call Gemini
	aiResponse, err := callGeminiWithSDK(prompt)
	if err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "AI failed: " + err.Error()})
		return
	}

	// 4. Parse AI Response
	var parsedResult AITaskResponse
	if err := json.Unmarshal([]byte(aiResponse), &parsedResult); err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "AI returned invalid format", "raw": aiResponse})
		return
	}

	// 5. Find Assignee
	var assigneeID primitive.ObjectID
	for _, u := range users {
		if u.Name == parsedResult.Assignee {
			assigneeID = u.ID
			break
		}
	}

	// 6. Save Task
	task := &models.Task{
		OrganizationID: orgID,
		ProjectID:      projectID,
		Title:          parsedResult.Title,
		Priority:       parsedResult.Priority,
		AssigneeID:     assigneeID,
		Status:         models.StatusTodo,
		CreatedBy:      userID,
		Subtasks:       []models.Subtask{},
	}

	if err := a.TaskRepo.Create(c.Request.Context(), task); err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to save task"})
		return
	}

	c.JSON(http.StatusOK, task)
}

	// 2. Build Context String

	contextStr += "PROJECTS:\n"
	for _, p := range projects {
		contextStr += fmt.Sprintf("- %s (Key: %s): %s\n", p.Name, p.Key, p.Description)
	}

	contextStr += "\nTEAM MEMBERS:\n"
	for _, u := range users {
		contextStr += fmt.Sprintf("- %s (%s)\n", u.Name, u.Role)
	}

	contextStr += "\nTASKS:\n"
	// Just iterate over the slice directly
	if tasksResult != nil {
		for _, t := range tasksResult {
			assigneeName := "Unassigned"
			for _, u := range users {
				if u.ID == t.AssigneeID {
					assigneeName = u.Name
					break
				}
			}
			contextStr += fmt.Sprintf("- [%s] %s | Status: %s | Priority: %s | Assignee: %s\n", t.ID.Hex(), t.Title, t.Status, t.Priority, assigneeName)
		}
	}

	// 3. Build Prompt
	prompt := fmt.Sprintf(`
		You are a helpful AI assistant for a project management app.
		Here is the current data of the workspace:
		%s
		
		User Question: "%s"
		
		Answer the user's question based ONLY on the data provided above. Be concise and professional. If the answer is not in the data, say "I don't have that information in the workspace."
	`, contextStr, in.Question)

	// 4. Call Gemini
	answer, err := callGeminiWithSDK(prompt)
	if err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"error": "AI is busy, please try again."})
		return
	}

	c.JSON(http.StatusOK, gin.H{"answer": answer})
}

// --- Gemini API Helper ---

func callGeminiWithSDK(prompt string) (string, error) {
	apiKey := os.Getenv("GEMINI_API_KEY")
	if apiKey == "" {
		return "", fmt.Errorf("GEMINI_API_KEY is missing in .env")
	}

	// Use Background context (no client deadline)
	ctx := context.Background()

	client, err := genai.NewClient(ctx, &genai.ClientConfig{
		APIKey:  apiKey,
		Backend: genai.BackendGeminiAPI,
	})
	if err != nil {
		return "", fmt.Errorf("client init failed: %v", err)
	}

	models := []string{"gemini-flash-latest", "gemini-2.5-flash", "gemini-3.8-flash"}
	var lastErr error

	for _, modelName := range models {
		fmt.Printf("ðŸ”„ Trying model: %s\n", modelName)

		resp, err := client.Models.GenerateContent(ctx, modelName,
			[]*genai.Content{
				{Parts: []*genai.Part{{Text: prompt}}},
			}, nil)

		if err == nil && len(resp.Candidates) > 0 && len(resp.Candidates[0].Content.Parts) > 0 {
			text := resp.Candidates[0].Content.Parts[0].Text
			fmt.Printf("âœ… Success with model: %s\n", modelName)
			return cleanJSON(text), nil
		}

		lastErr = err
		fmt.Printf("âš ï¸ Model %s failed: %v\n", modelName, err)
	}

	return "", fmt.Errorf("all AI models failed. Last error: %v", lastErr)
}

func cleanJSON(s string) string {
	if len(s) > 7 && s[:7] == "```json" {
		s = s[7:]
	}
	if len(s) > 3 && s[len(s)-3:] == "```" {
		s = s[:len(s)-3]
	}
	return s
}

	}

		ctx.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})

	}

		Details:        "Added a comment",
	})

	// 2. Send Notification to Assignee
	task, _ := c.TaskRepo.FindByID(ctx.Request.Context(), orgID, taskID)
	if task != nil && !task.AssigneeID.IsZero() && task.AssigneeID != userID {
		c.NotificationRepo.Create(ctx.Request.Context(), &models.Notification{
			OrganizationID: orgID,
			UserID:         task.AssigneeID,
			Title:          "New Comment",
			Message:        "Someone commented on your task: " + task.Title,
		})
	}

	// 3. Broadcast to WebSocket clients in the same Org
	// 3. Broadcast to WebSocket clients in the same Org (Run in background)
	broadcastMsg := fmt.Sprintf(`{"type":"NEW_COMMENT","orgId":"%s","taskId":"%s"}`, orgID.Hex(), taskID.Hex())
	go func() {
		ws.GlobalHub.Broadcast <- []byte(broadcastMsg)
	}()

	// 4. Send the final HTTP response ONLY ONCE
	ctx.JSON(http.StatusCreated, comment)

	"net/http"

	"github.com/gin-gonic/gin"
	"github.com/gorilla/websocket"
	"github.com/yourname/multi-tenant-pm/utils"
	ws "github.com/yourname/multi-tenant-pm/websocket"
)

var upgrader = websocket.Upgrader{
	CheckOrigin: func(r *http.Request) bool {
		return true // In production, restrict this to your frontend URL
	},
}

func HandleWS(c *gin.Context) {
	// Get token from query param (since browsers can't easily set headers for WS)
	token := c.Query("token")
	if token == "" {
		c.JSON(http.StatusUnauthorized, gin.H{"error": "Missing token"})
		return
	}

	// Parse token to get OrgID
	claims, err := utils.ParseToken(token)
	if err != nil {
		c.JSON(http.StatusUnauthorized, gin.H{"error": "Invalid token"})
		return
	}

	conn, err := upgrader.Upgrade(c.Writer, c.Request, nil)
	if err != nil {
		return
	}

	client := &ws.Client{
		Hub:   ws.GlobalHub,
		Conn:  conn,
		Send:  make(chan []byte, 256),
		OrgID: claims.OrganizationID,
	}

	client.Hub.Register <- client

	go client.WritePump()
	go client.ReadPump()
}

func AuthRequired() gin.HandlerFunc {
	return func(c *gin.Context) {
		h := c.GetHeader("Authorization")
		if !strings.HasPrefix(h, "Bearer ") {
			c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "missing token"})
			return
		}
		claims, err := utils.ParseToken(strings.TrimPrefix(h, "Bearer "))
		if err != nil {
			c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "invalid token"})
			return
		}
		c.Set("userId", claims.UserID)
		c.Set("orgId", claims.OrganizationID)
		c.Set("orgName", claims.OrgName) // <-- Added
		c.Set("role", claims.Role)
		c.Next()
	}
}

func (r *TaskRepo) AddSubtask(ctx context.Context, orgID, taskID primitive.ObjectID, subtask models.Subtask) error {
	filter := bson.M{"_id": taskID, "organizationId": orgID}

	// If subtasks is null or doesn't exist, $push will fail.
	// We use an aggregation pipeline to check and initialize the array first.
	update := bson.A{
		bson.M{
			"$set": bson.M{
				"subtasks": bson.M{
					"$cond": bson.M{
						"if":   bson.M{"$isArray": "$subtasks"},
						"then": "$subtasks",
						"else": bson.A{},
					},
				},
			},
		},
		bson.M{
			"$set": bson.M{
				"subtasks": bson.M{
					"$concatArrays": []interface{}{
						"$subtasks",
						bson.A{subtask},
					},
				},
			},
		},
	}

	_, err := r.coll().UpdateOne(ctx, filter, update)
	return err
}

	}

	_, err := r.coll().UpdateOne(ctx, filter, update)
	return err

type AuthService struct {
	Users *repositories.UserRepo
}

type RegisterInput struct {
	OrgName  string `json:"orgName" validate:"required,min=2"`
	Slug     string `json:"slug" validate:"required,min=2"`
	Name     string `json:"name" validate:"required"`
	Email    string `json:"email" validate:"required,email"`
	Password string `json:"password" validate:"required,min=6"`
}

type LoginInput struct {
	Email    string `json:"email" validate:"required,email"`
	Password string `json:"password" validate:"required"`
}

func (s *AuthService) Register(ctx context.Context, in RegisterInput) (string, *models.User, error) {
	existing, err := s.Users.FindByEmail(ctx, strings.ToLower(in.Email))
	if err != nil {
		return "", nil, err
	}
	if existing != nil {
		return "", nil, errors.New("email already registered")
	}

	// 1) Create org
	org := models.Organization{
		Name:      in.OrgName,
		Slug:      strings.ToLower(in.Slug),
		CreatedAt: time.Now(),
	}
	res, err := config.Database.Collection("organizations").InsertOne(ctx, org)
	if err != nil {
		return "", nil, err
	}
	org.ID = res.InsertedID.(primitive.ObjectID)

	// 2) Hash pw
	hash, err := utils.HashPassword(in.Password)
	if err != nil {
		return "", nil, err
	}

	// 3) Create owner user
	user := &models.User{
		OrganizationID: org.ID,
		Name:           in.Name,
		Email:          strings.ToLower(in.Email),
		PasswordHash:   hash,
		Role:           models.RoleOwner,
	}
	if err := s.Users.Create(ctx, user); err != nil {
		return "", nil, err
	}

	// 4) Generate Token with orgName
	token, err := utils.GenerateToken(user.ID.Hex(), org.ID.Hex(), org.Name, user.Role)
	return token, user, err
}

func (s *AuthService) Login(ctx context.Context, in LoginInput) (string, *models.User, error) {
	user, err := s.Users.FindByEmail(ctx, strings.ToLower(in.Email))
	if err != nil {
		return "", nil, err
	}
	if user == nil || !utils.CheckPassword(user.PasswordHash, in.Password) {
		return "", nil, errors.New("invalid credentials")
	}

	// Fetch organization name for the token
	var org models.Organization
	config.Database.Collection("organizations").FindOne(ctx, bson.M{"_id": user.OrganizationID}).Decode(&org)

	// Generate Token with orgName
	token, err := utils.GenerateToken(user.ID.Hex(), user.OrganizationID.Hex(), org.Name, user.Role)

func GenerateToken(userID, orgID, orgName, role string) (string, error) {
	hours, _ := strconv.Atoi(os.Getenv("JWT_EXPIRY_HOURS"))
	if hours == 0 {
		hours = 72
	}
	claims := Claims{
		UserID:         userID,
		OrganizationID: orgID,
		OrgName:        orgName, // <-- Added
		Role:           role,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(time.Now().Add(time.Duration(hours) * time.Hour)),
			IssuedAt:  jwt.NewNumericDate(time.Now()),
		},
	}
	return jwt.NewWithClaims(jwt.SigningMethodHS256, claims).
		SignedString([]byte(os.Getenv("JWT_SECRET")))
}

func ParseToken(tokenStr string) (*Claims, error) {
	tok, err := jwt.ParseWithClaims(tokenStr, &Claims{}, func(t *jwt.Token) (interface{}, error) {
		if _, ok := t.Method.(*jwt.SigningMethodHMAC); !ok {
			return nil, errors.New("unexpected signing method")
		}
		return []byte(os.Getenv("JWT_SECRET")), nil
	})
	if err != nil {
		return nil, err
	}
	claims, ok := tok.Claims.(*Claims)
	if !ok || !tok.Valid {
		return nil, errors.New("invalid token")
	}
	return claims, nil
}

	"log"
	"time"

	"github.com/gorilla/websocket"
)

const (
	writeWait      = 10 * time.Second
	pongWait       = 60 * time.Second
	pingPeriod     = (pongWait * 9) / 10
	maxMessageSize = 512
)

type Client struct {
	Hub    *Hub
	Conn   *websocket.Conn
	Send   chan []byte
	OrgID  string // Used to broadcast only to the same company
}

func (c *Client) ReadPump() {
	defer func() {
		c.Hub.Unregister <- c
		c.Conn.Close()
	}()
	c.Conn.SetReadLimit(maxMessageSize)
	c.Conn.SetReadDeadline(time.Now().Add(pongWait))
	c.Conn.SetPongHandler(func(string) error { c.Conn.SetReadDeadline(time.Now().Add(pongWait)); return nil })
	
	for {
		_, _, err := c.Conn.ReadMessage()
		if err != nil {
			if websocket.IsUnexpectedCloseError(err, websocket.CloseGoingAway, websocket.CloseAbnormalClosure) {
				log.Printf("error: %v", err)
			}
			break
		}
	}
}

func (c *Client) WritePump() {
	ticker := time.NewTicker(pingPeriod)
	defer func() {
		ticker.Stop()
		c.Conn.Close()
	}()
	for {
		select {
		case message, ok := <-c.Send:
			c.Conn.SetWriteDeadline(time.Now().Add(writeWait))
			if !ok {
				c.Conn.WriteMessage(websocket.CloseMessage, []byte{})
				return
			}
			w, err := c.Conn.NextWriter(websocket.TextMessage)
			if err != nil {
				return
			}
			w.Write(message)
			if err := w.Close(); err != nil {
				return
			}
		case <-ticker.C:
			c.Conn.SetWriteDeadline(time.Now().Add(writeWait))
			if err := c.Conn.WriteMessage(websocket.PingMessage, nil); err != nil {
				return
			}
		}
	}
}

		select {
		case client := <-h.Register:
			h.mu.Lock()
			h.Clients[client] = true
			h.mu.Unlock()
			log.Println("âœ… WebSocket Client Connected")
		case client := <-h.Unregister:
			h.mu.Lock()
			if _, ok := h.Clients[client]; ok {
				delete(h.Clients, client)
				close(client.Send)
			}
			h.mu.Unlock()
			log.Println("âŒ WebSocket Client Disconnected")
		case message := <-h.Broadcast:
			h.mu.Lock()
			for client := range h.Clients {
				select {
				case client.Send <- message:
				default:
					close(client.Send)
					delete(h.Clients, client)
				}
			}
			h.mu.Unlock()
		}

    if (!isAuthenticated || !token) {

    }

    // 1. Determine the correct WebSocket URL
    // If we are in production, replace http/https with ws/wss
    const apiUrl = import.meta.env.VITE_API_URL || "http://localhost:8080/api";
    const wsUrl = apiUrl.replace(/^http/, "ws"); // turns http:// into ws://, https:// into wss://

    // 2. Connect
    ws.current = new WebSocket(`${wsUrl}/ws?token=${token}`);

    ws.current.onopen = () => {
      console.log("âœ… WebSocket Connected");
    };

    ws.current.onmessage = (event) => {
      const data = JSON.parse(event.data);
      if (data.type === "NEW_COMMENT") {
        console.log("ðŸ“© Real-time update received!");

        queryClient.invalidateQueries({ queryKey: ["comments", data.taskId] });
        queryClient.invalidateQueries({
          queryKey: ["activities", data.taskId],
        });
      }
    };

    ws.current.onclose = () => {
      console.log("âŒ WebSocket Disconnected");
    };

    return () => {
      if (ws.current) ws.current.close();
    };
  }, [isAuthenticated, token, queryClient]);

import { createSlice } from '@reduxjs/toolkit';

// Check if user is already logged in (from previous session)
const savedAuth = JSON.parse(localStorage.getItem('auth') || 'null');

const initialState = {
    user: savedAuth ? savedAuth.user : null,
    token: savedAuth ? savedAuth.token : null,
    isAuthenticated: savedAuth ? true : false,
};

const authSlice = createSlice({
    name: 'auth',
    initialState,
    reducers: {
        // Call this after successful login/register
        setCredentials: (state, action) => {
            const { user, token } = action.payload;
            state.user = user;
            state.token = token;
            state.isAuthenticated = true;

            // Save to localStorage so user stays logged in on refresh
            localStorage.setItem('auth', JSON.stringify({ user, token }));
        },

        // Call this to log out
        logout: (state) => {
            state.user = null;
            state.token = null;
            state.isAuthenticated = false;
            localStorage.removeItem('auth');
        },
    },
});

export const { setCredentials, logout } = authSlice.actions;
export default authSlice.reducer;
